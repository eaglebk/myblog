# Composition Basics: Rule of Thirds

> An overview of the fundamental rule of photography composition: how to balance your frame using a grid of thirds.

- **URL:** https://eagle2000.space/en/photography/composition-basics/
- **Date:** 2026-04-25
- **Categories:** photography
- **Tags:** composition, tips

---

<p>Composition is how the elements of a frame are arranged relative to each other. One of the simplest and most effective ways to improve your shots is to use the <strong>rule of thirds</strong>.</p>

<h2 class="relative group">What is the rule of thirds?
    <div id="what-is-the-rule-of-thirds" class="anchor"></div>
    
    <span
        class="absolute top-0 w-6 transition-opacity opacity-0 -start-6 not-prose group-hover:opacity-100 select-none">

    </span>
    
</h2>
<p>Imagine your frame is divided into a 3x3 grid by two horizontal and two vertical lines. The rule of thirds suggests placing important elements of the scene along these lines or at their intersections (active points).</p>

<h2 class="relative group">Intersections (Active Zones)
    <div id="intersections-active-zones" class="anchor"></div>
    
    <span
        class="absolute top-0 w-6 transition-opacity opacity-0 -start-6 not-prose group-hover:opacity-100 select-none">

    </span>
    
</h2>
<p>The four intersection points of the grid lines are considered areas to which the human eye naturally pays attention first:</p>
<ul>
<li><strong>Portraits</strong>: Place the subject&rsquo;s eyes on one of the top intersection points.</li>
<li><strong>Landscapes</strong>: Position the horizon line along either the bottom or top grid line (depending on whether you want to emphasize the sky or the ground).</li>
</ul>

<h2 class="relative group">Why does it work?
    <div id="why-does-it-work" class="anchor"></div>
    
    <span
        class="absolute top-0 w-6 transition-opacity opacity-0 -start-6 not-prose group-hover:opacity-100 select-none">
        <a class="text-primary-300 dark:text-neutral-700 !no-underline" href="#why-does-it-work" aria-label="Anchor">#</a>
    </span>
    
</h2>
<p>Unlike centered composition, which often feels static and overly formal, shifting the main subject according to the rule of thirds creates a sense of dynamic flow and space. It encourages the viewer&rsquo;s eye to wander around the frame, exploring the details.</p>

  (function () {
    function initGame() {
      const wrapper = document.querySelector(".thirds-game-wrapper");
      if (!wrapper) return;

      const isEn =  true ;
      const polaroidText = 'Seascape, 2026';

      const viewport = wrapper.querySelector(".camera-viewport-container");
      const image = wrapper.querySelector(".camera-panorama-image");
      const gridOverlay = wrapper.querySelector(".camera-grid-overlay");
      const shutterFlash = wrapper.querySelector(".camera-shutter-flash");
      const resultsPanel = wrapper.querySelector(".camera-results-panel");
      const resultsCard = wrapper.querySelector(".results-card");
      const btnToggleGrid = wrapper.querySelector(".btn-toggle-grid");
      const gridStatusText = wrapper.querySelector(".grid-status-text");
      const btnShutter = wrapper.querySelector(".btn-shutter");
      const btnRetry = wrapper.querySelector(".btn-retry");
      const btnDownload = wrapper.querySelector(".btn-download");
      const inputCaption = wrapper.querySelector(".input-caption");
      const btnCloseResults = wrapper.querySelector(".btn-close-results");

      const scoreEmoji = wrapper.querySelector(".score-emoji");
      const scoreBarFill = wrapper.querySelector(".score-bar-fill");
      const scoreValue = wrapper.querySelector(".score-value");
      const scoreTextValue = wrapper.querySelector(".score-text-value");
      const feedbackText = wrapper.querySelector(".feedback-text");

      
      const TARGETS = [
        
        { name: "Lighthouse", nameAcc: "lighthouse", verb: "is positioned", xPct: 0.505, yPct: 0.580 },
        { name: "Sailboat", nameAcc: "sailboat", verb: "is positioned", xPct: 0.245, yPct: 0.730 }
        
      ];

      const imgWidth = 2000;
      const imgHeight = 2000;

      let vWidth = 0, vHeight = 0;
      let currentScale = 1;
      let currentX = 0, currentY = 0;

      let isDragging = false;
      let startX = 0, startY = 0;
      let isFrozen = false;
      let latestScore = 0;

      function boundPan() {
        vWidth = viewport.clientWidth;
        vHeight = viewport.clientHeight;
        if (vWidth === 0) return;

        const minScale = Math.max(vWidth / imgWidth, vHeight / imgHeight);
        if (currentScale < minScale) currentScale = minScale;
        if (currentScale > 3.0) currentScale = 3.0;

        const minX = vWidth - (imgWidth * currentScale);
        const minY = vHeight - (imgHeight * currentScale);

        if (currentX > 0) currentX = 0;
        if (currentX < minX) currentX = minX;

        if (currentY > 0) currentY = 0;
        if (currentY < minY) currentY = minY;
      }

      function updateTransform() {
        image.style.transform = `translate3d(${currentX}px, ${currentY}px, 0) scale(${currentScale})`;
      }

      function resetView() {
        vWidth = viewport.clientWidth;
        vHeight = viewport.clientHeight;
        if (vWidth === 0) return;

        const minScale = Math.max(vWidth / imgWidth, vHeight / imgHeight);
        currentScale = minScale * 1.35;

        currentX = -(imgWidth * currentScale - vWidth) / 2;
        currentY = -(imgHeight * currentScale - vHeight) / 2;

        boundPan();
        updateTransform();
      }

      let isGridOn = true;
      btnToggleGrid.addEventListener("click", () => {
        isGridOn = !isGridOn;
        gridOverlay.style.opacity = isGridOn ? "1" : "0";
        gridStatusText.textContent = isGridOn ? (isEn ? "ON" : "ВКЛ") : (isEn ? "OFF" : "ВЫКЛ");
      });

      viewport.addEventListener("mousedown", (e) => {
        if (isFrozen) return;
        isDragging = true;
        image.style.cursor = "grabbing";
        startX = e.clientX - currentX;
        startY = e.clientY - currentY;
      });

      window.addEventListener("mousemove", (e) => {
        if (!isDragging || isFrozen) return;
        currentX = e.clientX - startX;
        currentY = e.clientY - startY;
        boundPan();
        updateTransform();
      });

      window.addEventListener("mouseup", () => {
        isDragging = false;
        image.style.cursor = "grab";
      });

      window.addEventListener("mouseleave", () => {
        isDragging = false;
        image.style.cursor = "grab";
      });

      viewport.addEventListener("wheel", (e) => {
        if (isFrozen) return;
        e.preventDefault();

        const rect = viewport.getBoundingClientRect();
        const mouseX = e.clientX - rect.left;
        const mouseY = e.clientY - rect.top;

        const imageX = (mouseX - currentX) / currentScale;
        const imageY = (mouseY - currentY) / currentScale;

        const zoomFactor = 0.04;
        let nextScale = currentScale + (e.deltaY < 0 ? zoomFactor : -zoomFactor);

        currentScale = nextScale;
        boundPan();

        currentX = mouseX - imageX * currentScale;
        currentY = mouseY - imageY * currentScale;

        boundPan();
        updateTransform();
      }, { passive: false });

      let startPinchDist = 0;
      let startPinchScale = 1;
      let pinchImageX = 0, pinchImageY = 0;
      let isPinching = false;

      viewport.addEventListener("touchstart", (e) => {
        if (isFrozen) return;
        const rect = viewport.getBoundingClientRect();

        if (e.touches.length === 1) {
          isDragging = true;
          isPinching = false;
          startX = e.touches[0].clientX - currentX;
          startY = e.touches[0].clientY - currentY;
        } else if (e.touches.length === 2) {
          isDragging = false;
          isPinching = true;

          const x1 = e.touches[0].clientX;
          const y1 = e.touches[0].clientY;
          const x2 = e.touches[1].clientX;
          const y2 = e.touches[1].clientY;

          startPinchDist = Math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2);
          startPinchScale = currentScale;

          const midX = (x1 + x2) / 2 - rect.left;
          const midY = (y1 + y2) / 2 - rect.top;

          pinchImageX = (midX - currentX) / currentScale;
          pinchImageY = (midY - currentY) / currentScale;
        }
      }, { passive: true });

      viewport.addEventListener("touchmove", (e) => {
        if (isFrozen) return;
        const rect = viewport.getBoundingClientRect();

        if (e.touches.length === 1 && isDragging) {
          currentX = e.touches[0].clientX - startX;
          currentY = e.touches[0].clientY - startY;
          boundPan();
          updateTransform();
        } else if (e.touches.length === 2 && isPinching) {
          const x1 = e.touches[0].clientX;
          const y1 = e.touches[0].clientY;
          const x2 = e.touches[1].clientX;
          const y2 = e.touches[1].clientY;

          const currentDist = Math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2);
          if (startPinchDist === 0) return;

          currentScale = startPinchScale * (currentDist / startPinchDist);
          boundPan();

          const midX = (x1 + x2) / 2 - rect.left;
          const midY = (y1 + y2) / 2 - rect.top;

          currentX = midX - pinchImageX * currentScale;
          currentY = midY - pinchImageY * currentScale;

          boundPan();
          updateTransform();
        }
      }, { passive: true });

      viewport.addEventListener("touchend", () => {
        isDragging = false;
        isPinching = false;
      });

      btnShutter.addEventListener("click", () => {
        if (isFrozen) return;
        isFrozen = true;

        shutterFlash.style.opacity = "1";
        setTimeout(() => shutterFlash.style.opacity = "0", 100);

        vWidth = viewport.clientWidth;
        vHeight = viewport.clientHeight;

        const intersections = [
          { x: vWidth / 3, y: vHeight / 3 },
          { x: (2 * vWidth) / 3, y: vHeight / 3 },
          { x: vWidth / 3, y: (2 * vHeight) / 3 },
          { x: (2 * vWidth) / 3, y: (2 * vHeight) / 3 }
        ];

        let minDistance = Infinity;
        let bestScore = 0;
        let targetObj = TARGETS[0];

        TARGETS.forEach(target => {
          const vTargetX = (target.xPct * imgWidth * currentScale) + currentX;
          const vTargetY = (target.yPct * imgHeight * currentScale) + currentY;

          intersections.forEach(pt => {
            const dx = vTargetX - pt.x;
            const dy = vTargetY - pt.y;
            const distance = Math.sqrt(dx * dx + dy * dy);

            if (distance < minDistance) {
              minDistance = distance;
              targetObj = target;

              let score = 0;
              if (distance <= 30) {
                score = 100;
              } else if (distance < 110) {
                score = Math.round(100 - ((distance - 30) / 80) * 100);
              }
              bestScore = score;
            }
          });
        });

        scoreValue.textContent = bestScore;
        scoreBarFill.style.width = `${bestScore}%`;
        latestScore = bestScore;

        if (bestScore >= 85) {
          scoreEmoji.textContent = "🏆";
          scoreBarFill.className = "score-bar-fill h-full bg-emerald-500 transition-all duration-700 ease-out";
          scoreTextValue.className = "text-xl font-black text-emerald-400 m-0 score-text-value";
          feedbackText.innerHTML = isEn
            ? `Great shot! You caught the <b>${targetObj.nameAcc}</b> right at the composition power point. Perfect rule of thirds alignment!`
            : `Отличный кадр! Вы поймали <b>${targetObj.nameAcc}</b> точно в активную точку силы. Идеальное следование правилу третей!`;
        } else if (bestScore >= 50) {
          scoreEmoji.textContent = "👍";
          scoreBarFill.className = "score-bar-fill h-full bg-amber-500 transition-all duration-700 ease-out";
          scoreTextValue.className = "text-xl font-black text-amber-400 m-0 score-text-value";
          feedbackText.innerHTML = isEn
            ? `Nice shot! The <b>${targetObj.name}</b> ${targetObj.verb} close to a grid intersection. The composition looks balanced.`
            : `Хороший кадр! <b>${targetObj.name}</b> ${targetObj.verb} близко к пересечению сетки. Композиция выглядит сбалансировано.`;
        } else {
          scoreEmoji.textContent = "📷";
          scoreBarFill.className = "score-bar-fill h-full bg-red-500 transition-all duration-700 ease-out";
          scoreTextValue.className = "text-xl font-black text-red-400 m-0 score-text-value";
          feedbackText.innerHTML = isEn
            ? `The elements in the frame are far from the power points. Try aligning a grid intersection over the <b>lighthouse</b> or the <b>sailboat</b>.`
            : `Объекты кадра далеко от активных точек. Наведите перекрестие сетки на <b>маяк</b> или <b>лодку</b>.`;
        }

        setTimeout(() => {
          resultsPanel.classList.add("opacity-100");
          resultsPanel.style.pointerEvents = "auto";
          resultsCard.style.transform = "scale(1)";
        }, 200);
      });

      
      btnRetry.addEventListener("click", () => {
        isFrozen = false;
        resultsPanel.classList.remove("opacity-100");
        resultsPanel.style.pointerEvents = "none";
        resultsCard.style.transform = "scale(0.95)";
        scoreBarFill.style.width = "0%";
        inputCaption.value = "";
        resetView();
      });

      
      if (btnCloseResults) {
        btnCloseResults.addEventListener("click", () => {
          isFrozen = false;
          resultsPanel.classList.remove("opacity-100");
          resultsPanel.style.pointerEvents = "none";
          resultsCard.style.transform = "scale(0.95)";
          scoreBarFill.style.width = "0%";
        });
      }

      btnDownload.addEventListener("click", () => {
        const imgUrl = image.style.backgroundImage.replace(/url\(['"]?(.*?)['"]?\)/, '$1');
        const imgObj = new Image();
        imgObj.onload = function () {
          const canvas = document.createElement("canvas");

          const factorX = imgObj.naturalWidth / imgWidth;
          const factorY = imgObj.naturalHeight / imgHeight;

          const sx = (-currentX / currentScale) * factorX;
          const sy = (-currentY / currentScale) * factorY;
          const sw = (vWidth / currentScale) * factorX;
          const sh = (vHeight / currentScale) * factorY;

          const margin = Math.round(sw * 0.08);
          const bottomMargin = Math.round(sh * 0.22);

          canvas.width = Math.round(sw) + 2 * margin;
          canvas.height = Math.round(sh) + margin + bottomMargin;

          const ctx = canvas.getContext("2d");

          ctx.fillStyle = "#ffffff";
          ctx.fillRect(0, 0, canvas.width, canvas.height);

          ctx.drawImage(imgObj, sx, sy, sw, sh, margin, margin, Math.round(sw), Math.round(sh));

          ctx.strokeStyle = "rgba(0, 0, 0, 0.12)";
          ctx.lineWidth = Math.max(1, Math.round(sw * 0.002));
          ctx.strokeRect(margin, margin, Math.round(sw), Math.round(sh));

          const text = inputCaption.value.trim() || polaroidText;
          const fontSize = Math.round(bottomMargin * 0.45);
          ctx.font = `bold ${fontSize}px 'Caveat', 'Bradley Hand', 'Segoe Print', cursive`;
          ctx.fillStyle = "#262626";
          ctx.textAlign = "center";
          ctx.textBaseline = "middle";

          const textX = canvas.width / 2;
          const textY = canvas.height - (bottomMargin / 2);
          ctx.fillText(text, textX, textY);

          const link = document.createElement("a");
          link.download = `polaroid_${latestScore}pct.jpg`;
          link.href = canvas.toDataURL("image/jpeg", 0.95);
          document.body.appendChild(link);
          link.click();
          document.body.removeChild(link);
        };
        imgObj.src = imgUrl;
      });

      setTimeout(resetView, 50);
      window.addEventListener("resize", resetView);
    }

    if (document.readyState === "complete" || document.readyState === "interactive") {
      initGame();
    } else {
      window.addEventListener("DOMContentLoaded", initGame);
    }
  })();
</script>
